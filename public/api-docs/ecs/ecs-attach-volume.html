const axios = require('axios');
let data = JSON.stringify({
  "subscription_name": "string",
  "resource_group_name": "string",
  "zone": "string",
  "ecs_instance_name": "string",
  "volume_uuid": "string",
  "volume_size": 0,
  "lifecycle": true
});

let config = {
  method: 'post',
maxBodyLength: Infinity,
  url: 'https://management.sensecoreapi.cn/compute/ecs/v1/subscriptions/:subscription_name/resourceGroups/:resource_group_name/zones/:zone/ecsInstances/undefined:attachVolume',
  headers: { 
    'Content-Type': 'application/json', 
    'Authorization': 'Bearer <TOKEN>'
  },
  data : data
};

axios(config)
.then((response) => {
  console.log(JSON.stringify(response.data));
})
.catch((error) => {
  console.log(error);
